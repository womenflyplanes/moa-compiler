def gamma_row(v,s):
    off=0
    for k in range(len(v)):
        prod=1
        for j in range(k+1,len(s)):
            prod*=s[j]
        off+=v[k]*prod
    return off

def gamma_col(v,s):
    off=0
    for k in range(len(v)):
        prod=1
        for j in range(0,k):
            prod*=s[j]
        off+=v[k]*prod
    return off
